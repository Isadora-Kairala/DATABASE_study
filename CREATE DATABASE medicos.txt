CREATE DATABASE medicos

USE medicos 

CREATE TABLE medico(
	codMed INT PRIMARY KEY IDENTITY(1,1),
	nome varchar (80),
	idade int,
	salarioo money
	)

CREATE TABLE especialidade(
	codEsp INT PRIMARY KEY IDENTITY(1,1),
	nome varchar(80)
	)
